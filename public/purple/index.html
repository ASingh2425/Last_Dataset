PYTHON STARTER CODE // 4-ZONE MERGE & CATBOOST PIPELINE

Use this script to merge the 4 fragments into a unified test set, clean missing values, train an ensemble model, and generate your submission file.

reconstruct_and_predict.py Python 3.10+ / pandas / catboost / scikit-learn 
import pandas as pd
import numpy as np
from catboost import CatBoostRegressor
from sklearn.linear_model import ElasticNet
from sklearn.metrics import r2_score

# ============================================================
# 1. LOAD & RECONSTRUCT TEST SET FROM 4 ZONE FRAGMENTS
# ============================================================
print("Reconstructing test.csv across 4 sector archives...")
comm = pd.read_csv("community_final.csv")
radio = pd.read_csv("radio_final.csv")
foren = pd.read_csv("forensics_final.csv")
res = pd.read_csv("research_final.csv")

# Inner join on Sanctuary_ID (120 survivors x 78 feature columns)
test = comm.merge(radio, on="Sanctuary_ID", how="inner") \
           .merge(foren, on="Sanctuary_ID", how="inner") \
           .merge(res, on="Sanctuary_ID", how="inner")

print(f"Test Set Reconstructed: {test.shape[0]} rows, {test.shape[1]} columns")

# ============================================================
# 2. LOAD TRAINING DATA
# ============================================================
train = pd.read_csv("train.csv")
TARGET = "Survival_Probability"
y_train = train[TARGET].values

# ============================================================
# 3. FEATURE SELECTION & PREPROCESSING
# ============================================================
drop_cols = ["Historical_ID", "Sanctuary_ID", TARGET, "Incident_Link_ID", 
             "Duty_Ref", "Vault_Case_ID", "Transfer_Manifest_ID", "Medical_Tag_ID"]

features = [c for c in train.columns if c not in drop_cols and c in test.columns]
X_train = train[features].copy()
X_test = test[features].copy()

# Categorical column handling
cat_cols = X_train.select_dtypes(include=["object"]).columns.tolist()
for c in cat_cols:
    X_train[c] = X_train[c].astype(str).fillna("Missing")
    X_test[c] = X_test[c].astype(str).fillna("Missing")

# Numerical column median imputation
num_cols = [c for c in features if c not in cat_cols]
for c in num_cols:
    med = X_train[c].median()
    X_train[c] = X_train[c].fillna(med)
    X_test[c] = X_test[c].fillna(med)

# ============================================================
# 4. TRAIN MODEL (CATBOOST REGRESSOR)
# ============================================================
print("Training CatBoost Survival Predictor...")
model = CatBoostRegressor(
    iterations=800,
    learning_rate=0.04,
    depth=6,
    cat_features=cat_cols,
    random_seed=42,
    verbose=100
)
model.fit(X_train, y_train)

# ============================================================
# 5. GENERATE PREDICTIONS & SUBMISSION CSV
# ============================================================
raw_preds = model.predict(X_test)
preds_clipped = np.clip(raw_preds, 0.0, 1.0)

submission = pd.DataFrame({
    "Sanctuary_ID": test["Sanctuary_ID"],
    "Survival_Probability": np.round(preds_clipped, 6)
})

submission.to_csv("my_submission.csv", index=False)
print("SUCCESS: 'my_submission.csv' generated with 120 prediction rows!")
print("Transmit 'my_submission.csv' in the SUBMIT PREDICTIONS tab.")
